\ejercicio{
    Escribe los siguientes números en la forma $x + iy$:
    \begin{itemize}
        \item[1)] $\frac{1}{i}$
        \item[2)] $\frac{3+2i}{1-i}$
        \item[3)] $(a+bi)^3$
    \end{itemize}
}{
    Resolvamos:
    \begin{itemize}
        \item[1)] $\frac{1}{i} = -i = 0 - 1i$.
        \item[2)] $\frac{3+2i}{1-i} = \frac{(3+2i)(1+i)}{(1-i)(1+i)} = \frac{3 + 5i - 2}{1+1} = \frac{1}{2} + \frac{5}{2} i$.
        \item[3)] $(a+bi)^3 = a^3 + 3a^2bi + 3ab^2i^2 + b^3i^3 = (a^3 - 3ab^2) + (3a^2b - b^3)i$.
    \end{itemize}
}

\ejercicio{
    Sea $P(z) = a_0 + a_1 z + a_2 z^2 + \cdots + a_n z^n$ con $a_i \in \mathbb{R} \quad \forall i \in \{0, 1, \ldots, n\}$. Prueba que si $z \in \mathbb{C}$ es raíz de $P$ (es decir, $P(z)=0$), entonces $\overline{z}$ también es raíz de $P$.
}{
    Sea $z \in \mathbb{C}$ tal que $P(z)=0$. Entonces
    \[
    P(z) = a_0 + a_1 z + a_2 z^2 + \cdots + a_n z^n = 0.
    \]

    Tomando conjugados complejos en ambos lados, obtenemos
    \[
    \overline{P(z)} = \overline{0} = 0.
    \]

    Usando la propiedad de que el conjugado de una suma es la suma de los conjugados y el conjugado de un producto es el producto de los conjugados, tenemos
    \[
    \overline{P(z)} = \overline{a_0} + \overline{a_1 z} + \overline{a_2 z^2} + \cdots + \overline{a_n z^n}.
    \]

    Dado que los coeficientes $a_i$ son reales, se cumple que $\overline{a_i} = a_i$. Por lo tanto,
    \[
    \overline{P(z)} = a_0 + a_1 \overline{z} + a_2 (\overline{z})^2 + \cdots + a_n (\overline{z})^n = P(\overline{z}).
    \]

    Así, hemos demostrado que
    \[
    P(\overline{z}) = 0,
    \]
    lo que implica que $\overline{z}$ también es raíz de $P$.
}

\ejercicio{
    Sea $z \in \mathbb{C}$ con $|z|<1$. Prueba que
    \[
    \left| \frac{z-a}{1-\overline{a}z} \right| < 1 
    \quad \text{si y sólo si } |a|<1,
    \]
    y que
    \[
    \left| \frac{z-a}{1-\overline{a}z} \right| = 1
    \quad \text{si y sólo si } |a|=1.
    \]
}{
    Consideremos
    \[
    w = \frac{z-a}{1-\overline{a}z}.
    \]
    Calculemos la diferencia entre el denominador y el numerador al cuadrado (módulo):

    \begin{align*}
    |1-\overline{a}z|^2 - |z-a|^2
    &= (1-\overline{a}z)(1-a\overline{z}) - (z-a)(\overline{z}-\overline{a}) \\[4pt]
    &= 1-a\overline{z}-\overline{a}z+|a|^2|z|^2
    - \big( |z|^2 - z\overline{a} - a\overline{z} + |a|^2 \big) \\[4pt]
    &= 1-|z|^2 -|a|^2 + |a|^2|z|^2 \\[4pt]
    &= (1-|z|^2)(1-|a|^2).
    \end{align*}

    Dividiendo por $|1-\overline{a}z|^2$ obtenemos la identidad fundamental:
    \[
    1 - |w|^2 \;=\; \frac{(1-|z|^2)(1-|a|^2)}{|1-\overline{a}z|^2}.
    \]

    De aquí se deduce:

    \begin{itemize}
        \item Si $|a|<1$ y $|z|<1$, entonces $(1-|z|^2)(1-|a|^2) > 0$, por lo que
        \[
        1-|w|^2 > 0 \quad \Longrightarrow \quad |w|<1.
        \]

        \item Si $|a|=1$, entonces $(1-|a|^2)=0$, y por tanto
        \[
        1-|w|^2 = 0 \quad \Longrightarrow \quad |w|=1.
        \]

        \item Si $|a|>1$ y $|z|<1$, entonces $(1-|a|^2)<0$, lo que implica
        \[
        1-|w|^2 < 0 \quad \Longrightarrow \quad |w|>1.
        \]
    \end{itemize}

    En conclusión:
    \[
    \left| \frac{z-a}{1-\overline{a}z} \right| < 1 
    \quad \Longleftrightarrow \quad |a|<1,
    \]
    y
    \[
    \left| \frac{z-a}{1-\overline{a}z} \right| = 1 
    \quad \Longleftrightarrow \quad |a|=1.
    \]

    \hfill $\Box$

    \[
        |\frac{z-a}{1-\overline{a}z}| < 1 \iff \frac{|z-a|}{|1-\overline{a}z|} < 1 \iff |z-a| < |1-\overline{a}z|
    \]
    \[
        \underbrace{\iff}_{|z-a|^2 \geq 0} |z-a|^2 < |1-\overline{a}z|^2 \iff (z-a)(\overline{z}-\overline{a}) < (1-\overline{a}z)(1-a\overline{z})
    \]
    \[
        \iff |z|^2 - z\overline{a} - a\overline{z} + |a|^2 < 1 - a\overline{z} - \overline{a}z + |a|^2|z|^2 \iff |z|^2 + |a|^2 < 1 + |a|^2|z|^2
    \]
    \[
        \iff |a|^2(1-|z|^2) < 1 - |z|^2 \underbrace{\iff}_{|z| < 1} |a|^2 < 1 \underbrace{\iff}_{|a| > 0} |a| < 1.
    \]

    De lo anterior se deduce inmediatamente que si $|a|<1$, entonces
    \[
    \left| \frac{z-a}{1-\overline{a}z} \right| < 1 \iff |z| < 1.
    \]
    y
    \[
    \left| \frac{z-a}{1-\overline{a}z} \right| = 1 \iff |z| = 1.
    \]
    Si $0 < |a| < 1$, y definimos la función
    \[
        \varphi_a(z) = \frac{z-a}{1-\overline{a}z}, \quad  \forall z \in \mathbb{C} \setminus \left\{\frac{1}{\overline{a}}\right\},
    \]
    el ejercicio nos dice que $\varphi_a(D(0,1)) \subset D(0,1)$ y $\varphi_a(\{z \in \mathbb{C} : |z|=1\}) \subset \{w \in \mathbb{C} : |w|=1\}$.

}

\ejercicio{
    Expresa $\cos(5\varphi)$ en función de potencias de $\sin \varphi$ y $\cos \varphi$.
}{
    Por la fórmula de De Moivre se tiene
    \[
    (\cos \varphi + i \sin \varphi)^5 = \cos(5\varphi) + i \sin(5\varphi).
    \]

    Expandimos el binomio:
    \[
    (\cos \varphi + i \sin \varphi)^5
    = \cos^5 \varphi 
    + 5i \cos^4 \varphi \sin \varphi
    - 10 \cos^3 \varphi \sin^2 \varphi
    - 10i \cos^2 \varphi \sin^3 \varphi
    + 5 \cos \varphi \sin^4 \varphi
    + i \sin^5 \varphi.
    \]

    Tomando la parte real, obtenemos
    \[
    \cos(5\varphi) 
    = \cos^5 \varphi - 10 \cos^3 \varphi \sin^2 \varphi + 5 \cos \varphi \sin^4 \varphi.
    \]

    \textbf{Expresión alternativa.}  
    Si sustituimos $\sin^2 \varphi = 1 - \cos^2 \varphi$, resulta un polinomio solo en $\cos \varphi$:
    \[
    \cos(5\varphi) = 16 \cos^5 \varphi - 20 \cos^3 \varphi + 5 \cos \varphi.
    \]

    \hfill $\Box$    
}

\ejercicio{
    Determina las raices cuartas de $i$.
}{
    Claramente $i$ tiene módulo $1$ y argumento $\frac{\pi}{2}$, por lo que sus raíces cuartas tienen módulo $\sqrt[4]{1} = 1$ y argumentos:
    \[
    \frac{\frac{\pi}{2} + 2k\pi}{4} = \frac{\pi}{8} + \frac{k\pi}{2}, \quad k=0,1,2,3.
    \]

    Por tanto, las raíces cuartas de $i$ son:
    \begin{enumerate}
        \item[1)] Para $k=0$: $\cos\left(\frac{\pi}{8}\right) + i \sin\left(\frac{\pi}{8}\right)$.
        \item[2)] Para $k=1$: $\cos\left(\frac{5\pi}{8}\right) + i \sin\left(\frac{5\pi}{8}\right)$.
        \item[3)] Para $k=2$: $\cos\left(\frac{9\pi}{8}\right) + i \sin\left(\frac{9\pi}{8}\right)$.
        \item[4)] Para $k=3$: $\cos\left(\frac{13\pi}{8}\right) + i \sin\left(\frac{13\pi}{8}\right)$.
    \end{enumerate}

    Estas son las cuatro raíces cuartas de $i$.
    
}

\ejercicio{
    Dado $z \in \mathbb{C} \setminus \{0\}$, determina por circunstanceas geometricas  los puntos:
    \begin{itemize}
        \item[1)] $z^2$
        \item[2)] $\frac{z}{\overline{z}}$
        \item[3)] $\frac{4+z}{z}$
    \end{itemize}
}{
    Sea $z = re^{i\theta}$ con $r>0$ y $\theta \in \mathbb{R}$. Entonces:
    \begin{itemize}
        \item[1)] $z^2 = r^2 e^{i 2\theta}$. Geométricamente, esto representa una dilatación del punto $z$ por un factor de $r$ (si $r>1$) o una contracción (si $0<r<1$), seguida de una rotación en sentido antihorario de un ángulo $\theta$ alrededor del origen.
        
        \item[2)] $\frac{z}{\overline{z}} = \frac{re^{i\theta}}{re^{-i\theta}} = e^{i 2\theta}$. Geométricamente, este punto se encuentra en el círculo unitario (ya que su módulo es 1) y representa una rotación en sentido antihorario de un ángulo $2\theta$ alrededor del origen.
        
        \item[3)] $\frac{4+z}{z} = \frac{4 + re^{i\theta}}{re^{i\theta}} = \frac{4}{re^{i\theta}} + 1 = \frac{4}{r}e^{-i\theta} + 1$. Geométricamente, esto representa una traslación del punto $\frac{4}{r}e^{-i\theta}$ (que está en la dirección opuesta a $z$ y a una distancia $\frac{4}{r}$ del origen) hacia la derecha por 1 unidad en el plano complejo.
    \end{itemize}
}

\ejercicio{
    Haya el modulo y argumento de:
    \begin{itemize}
        \item[1)] $(1+i)^5$
        \item[2)] Las raices septimas de $3+4i$
    \end{itemize}
}{
    \begin{itemize}
        \item[1)] Primero, expresamos $1+i$ en forma polar. El módulo es $\sqrt{1^2 + 1^2} = \sqrt{2}$ y el argumento es $\frac{\pi}{4}$. Por lo tanto, $(1+i)^5 = (\sqrt{2})^5 e^{i 5\frac{\pi}{4}} = 4\sqrt{2} e^{i \frac{5\pi}{4}}$. Así, el módulo es $4\sqrt{2}$ y el argumento es $\frac{5\pi}{4}$.
        
        \item[2)] Para encontrar las raíces séptimas de $3+4i$, primero expresamos $3+4i$ en forma polar. El módulo es $\sqrt{3^2 + 4^2} = 5$ y el argumento es $\tan^{-1}\left(\frac{4}{3}\right)$. Las raíces séptimas tienen módulo $\sqrt[7]{5}$ y argumentos dados por:
        \[
        \frac{\tan^{-1}\left(\frac{4}{3}\right) + 2k\pi}{7}, \quad k=0,1,2,3,4,5,6.
        \]
        Por lo tanto, las raíces séptimas de $3+4i$ tienen módulo $\sqrt[7]{5}$ y los argumentos indicados.
    \end{itemize}
}

\ejercicio{
    Sea $\{z_n\}_{n \in \mathbb{N}}$ una sucesión en $\mathbb{C}$ que converge a un punto $z \in \mathbb{C}$ si y sólo si las sucesiones $\{\Re(z_n)\}_{n \in \mathbb{N}}$ y $\{\Im(z_n)\}_{n \in \mathbb{N}}$ convergen a $\Re(z)$ e $\Im(z)$ respectivamente.
}{
    \begin{itemize}
        \item ($\Rightarrow$) 
        \[
            z_n \to z \iff |z_n - z| \to 0
        \]
        \[
            \begin{cases}
                0 \leq |\Re(z_n) - \Re(z)| \leq |\Re(z_n-z)| \leq |z_n - z| \quad \forall n \in \mathbb{N}  \\
                |z_n - z| \xrightarrow[n \to \infty]{} 0
            \end{cases}
        \]
        \[
        \implies |\Re(z_n) - \Re(z)| \xrightarrow[n \to \infty]{} 0 \iff \Re(z_n) \xrightarrow[n \to \infty]{} \Re(z)
        \]
        Análogamente, $\Im(z_n) \xrightarrow[n \to \infty]{} \Im(z)$.
        \item ($\Leftarrow$) Suponemos que 
        \[
            \begin{cases}
                \Re(z_n) \xrightarrow[n \to \infty]{} \Re(z) \\
                \Im(z_n) \xrightarrow[n \to \infty]{} \Im(z)
            \end{cases}
        \]
        entonces
        \[
            |z_n - z| = \sqrt{(\Re(z_n -z))^2 + (\Im(z_n - z))^2} \leq |\Re(z_n - z)| + |\Im(z_n - z)|
        \]
        \[
            = |\Re(z_n) - \Re(z)| + |\Im(z_n) - \Im(z)| \xrightarrow[n \to \infty]{} 0 + 0 = 0 
        \]
    \end{itemize}
}
